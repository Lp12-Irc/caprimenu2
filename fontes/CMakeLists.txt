add_custom_command(OUTPUT "${NATIVE_UPDATER}"
  COMMAND ${CMAKE_COMMAND} -E make_directory "${CMAKE_SOURCE_DIR}/assets/updater"
  COMMAND ${CMAKE_COMMAND} -E env "LIB=" "$ENV{WINDIR}/Microsoft.NET/Framework64/v4.0.30319/csc.exe"
    /nologo /target:winexe /optimize+ /utf8output "/out:${NATIVE_UPDATER_OUTPUT}"
    /r:System.Windows.Forms.dll /r:System.Drawing.dll /r:System.Net.Http.dll
    /r:System.IO.Compression.dll /r:System.IO.Compression.FileSystem.dll /r:System.Web.Extensions.dll
    "${NATIVE_UPDATER_SOURCE}"
  DEPENDS "${CMAKE_SOURCE_DIR}/updater/Program.cs" VERBATIM)
